get_filename_component(FIRMWARE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LVGL_ROOT "${FIRMWARE_ROOT}/managed_components/lvgl__lvgl")
if(NOT EXISTS "${LVGL_ROOT}/lvgl.h" OR NOT EXISTS "${FIRMWARE_ROOT}/build/config/sdkconfig.h")
    message(FATAL_ERROR "Run idf.py reconfigure in firmware first to resolve the pinned LVGL component and generate sdkconfig.h")
endif()
file(GLOB_RECURSE LVGL_SOURCES "${LVGL_ROOT}/src/*.c" "${LVGL_ROOT}/src/*.cpp")
add_library(lvgl_host STATIC ${LVGL_SOURCES})
target_include_directories(lvgl_host PUBLIC "${LVGL_ROOT}" "${FIRMWARE_ROOT}/build/config")
target_compile_definitions(lvgl_host PUBLIC LV_CONF_SKIP LV_CONF_KCONFIG_EXTERNAL_INCLUDE="sdkconfig.h")
add_executable(screen_tests screen_tests.cpp "${FIRMWARE_ROOT}/src/Screens.cpp"
    "${FIRMWARE_ROOT}/src/SpeedFont.c" "${FIRMWARE_ROOT}/src/DashFont.c")
target_include_directories(screen_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${FIRMWARE_ROOT}/include")
target_compile_features(screen_tests PRIVATE cxx_std_17)
target_compile_options(screen_tests PRIVATE -UNDEBUG)
target_link_libraries(screen_tests PRIVATE lvgl_host)
option(COMPANION_TEST_SANITIZERS "Instrument LVGL host render tests with address/undefined sanitizers" ON)
if(COMPANION_TEST_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target lvgl_host screen_tests)
        target_compile_options(${target} PRIVATE -g -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        # Pinned LVGL uses typed callbacks through generic callback typedefs.
        # Retain ASan and other UBSan checks; exclude that upstream convention.
        target_compile_options(lvgl_host PRIVATE -fno-sanitize=function)
    endif()
endif()
add_test(NAME firmware_lvgl_screens COMMAND screen_tests)
